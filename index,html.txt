    <script>
    const carrito = document.getElementById('lista-carrito');
    const totalCarrito = document.getElementById('total-carrito');
    const contadorCarrito = document.getElementById('contador-carrito');
    const botonVaciar = document.getElementById('vaciar-carrito');
    const botonFinalizar = document.getElementById('finalizar-compra');
    const inputBuscador = document.getElementById('buscador');
    
    const modalConfirmacion = document.getElementById('modal-confirmacion');
    const contenidoModalResumen = document.getElementById('contenido-modal-resumen');
    const botonConfirmarWhatsApp = document.getElementById('confirmar-enviar-whatsapp');
    const botonCerrarModal = document.getElementById('cerrar-modal');

    let articulosCarrito = [];
    let urlWhatsAppFinal = "";

    const botonesAgregar = document.querySelectorAll('.agregar-carrito');
    botonesAgregar.forEach(boton => {
        boton.addEventListener('click', (e) => {
            const productoHTML = e.target.closest('.producto');
            const id = productoHTML.getAttribute('data-id');
            const stockMaximo = parseInt(productoHTML.getAttribute('data-stock'));

            const productoEnCarrito = articulosCarrito.find(p => p.id === id);
            const cantidadActualEnCarrito = productoEnCarrito ? productoEnCarrito.cantidad : 0;

            if (cantidadActualEnCarrito >= stockMaximo) {
                alert('¡Lo sentimos! Has alcanzado el límite de stock disponible para este producto.');
                return;
            }

            const infoProducto = {
                id: id,
                nombre: productoHTML.getAttribute('data-nombre'),
                precio: parseFloat(productoHTML.getAttribute('data-precio')),
                cantidad: 1
            };

            if (productoEnCarrito) {
                productoEnCarrito.cantidad++;
            } else {
                articulosCarrito.push(infoProducto);
            }

            actualizarVistaStockYCarrito();
        });
    });

    document.addEventListener('click', (e) => {
        if (e.target && e.target.classList.contains('sumar-cantidad')) {
            const id = e.target.getAttribute('data-id');
            const productoHTML = document.querySelector(`.producto[data-id="${id}"]`);
            const stockMaximo = parseInt(productoHTML.getAttribute('data-stock'));
            const productoEnCarrito = articulosCarrito.find(p => p.id === id);

            if (productoEnCarrito && productoEnCarrito.cantidad >= stockMaximo) {
                alert('No hay más unidades disponibles de este producto en el inventario.');
                return;
            }

            if (productoEnCarrito) {
                productoEnCarrito.cantidad++;
            }
            actualizarVistaStockYCarrito();
        }

        if (e.target && e.target.classList.contains('restar-cantidad')) {
            const id = e.target.getAttribute('data-id');
            const productoEnCarrito = articulosCarrito.find(p => p.id === id);

            if (productoEnCarrito) {
                productoEnCarrito.cantidad--;
                if (productoEnCarrito.cantidad <= 0) {
                    articulosCarrito = articulosCarrito.filter(p => p.id !== id);
                }
            }
            actualizarVistaStockYCarrito();
        }
    });

    botonVaciar.addEventListener('click', () => {
        articulosCarrito = [];
        document.getElementById('nombre-cliente').value = '';
        document.getElementById('telefono-cliente').value = '';
        document.getElementById('direccion-cliente').value = '';
        document.getElementById('metodo-pago').selectedIndex = 0;
        inputBuscador.value = '';
        filtrarProductos();
        actualizarVistaStockYCarrito();
    });

    botonFinalizar.addEventListener('click', () => {
        if (articulosCarrito.length === 0) {
            alert('El carrito está vacío.');
            return;
        }

        const nombreCliente = document.getElementById('nombre-cliente').value.trim();
        if (nombreCliente === '') {
            alert('Por favor, escribe tu nombre antes de finalizar la compra.');
            document.getElementById('nombre-cliente').focus();
            return;
        }

        const telefonoCliente = document.getElementById('telefono-cliente').value.trim();
        if (telefonoCliente === '') {
            alert('Por favor, escribe tu número de WhatsApp o teléfono.');
            document.getElementById('telefono-cliente').focus();
            return;
        }

        const direccionCliente = document.getElementById('direccion-cliente').value.trim();
        if (direccionCliente === '') {
            alert('Por favor, escribe tu dirección de entrega.');
            document.getElementById('direccion-cliente').focus();
            return;
        }

        const metodoPago = document.getElementById('metodo-pago').value;

        let mensaje = "¡Hola! 👋 Mi nombre es *" + nombreCliente + "* y deseo realizar el siguiente pedido en *Tienda JR*:\n\n";
        let resumenHTML = "<strong>Cliente:</strong> " + nombreCliente + "<br><strong>Teléfono:</strong> " + telefonoCliente + "<br><strong>Dirección:</strong> " + direccionCliente + "<br><strong>Pago:</strong> " + metodoPago + "<br><br><strong>Productos:</strong><br>";
        
        let total = 0;
        articulosCarrito.forEach(item => {
            const subtotal = item.precio * item.cantidad;
            total += subtotal;
            mensaje += "▪️ " + item.cantidad + "x " + item.nombre + " - $" + subtotal.toFixed(2) + "\n";
            resumenHTML += "▪️ " + item.cantidad + "x " + item.nombre + " ($" + subtotal.toFixed(2) + ")<br>";
        });

        mensaje += "\n*Total a pagar: $" + total.toFixed(2) + "*\n";

    function actualizarVistaStockYCarrito() {
        carrito.innerHTML = '';
        let total = 0;
        let totalProductos = 0;

        articulosCarrito.forEach(producto => {
            const li = document.createElement('li');
            li.style.display = 'flex';
            li.style.justifyContent = 'space-between';
            li.style.alignItems = 'center';

            li.innerHTML = 
                '<span>' + producto.nombre + ' ($' + producto.precio + ')</span>' +
                '<div>' +
                    '<button class="btn-cantidad restar-cantidad" data-id="' + producto.id + '">-</button>' +
                    '<span>' + producto.cantidad + '</span>' +
                    '<button class="btn-cantidad sumar-cantidad" data-id="' + producto.id + '">+</button>' +
                '</div>';
            carrito.appendChild(li);

            total += producto.precio * producto.cantidad;
            totalProductos += producto.cantidad;
        });

        totalCarrito.textContent = total.toFixed(2);
        contadorCarrito.textContent = totalProductos;

        const productosHTML = document.querySelectorAll('.producto');
        productosHTML.forEach(prod => {
            const id = prod.getAttribute('data-id');
            const stockMaximo = parseInt(prod.getAttribute('data-stock'));
            const etiqueta = prod.querySelector('.etiqueta-stock');
            const botonAgregar = prod.querySelector('.agregar-carrito');
            const esOferta = prod.classList.contains('oferta');

            const productoEnCarrito = articulosCarrito.find(p => p.id === id);
            const cantidadComprada = productoEnCarrito ? productoEnCarrito.cantidad : 0;
            const stockRestante = stockMaximo - cantidadComprada;

            if (stockRestante <= 0) {
                etiqueta.className = 'etiqueta-stock stock-agotado';
                etiqueta.textContent = '❌ Agotado';
                botonAgregar.disabled = true;
                botonAgregar.textContent = 'Agotado';
            } else {
                botonAgregar.disabled = false;
                botonAgregar.textContent = 'Agregar';
                if (esOferta) {
                    etiqueta.className = 'etiqueta-stock etiqueta-oferta';
                    // Mantiene un texto personalizado de descuento si hay stock
                } else if (stockRestante <= 1) {
                    etiqueta.className = 'etiqueta-stock stock-limitado';
                    etiqueta.textContent = '🔥 ¡Última unidad! (1)';
                } else {
                    etiqueta.className = 'etiqueta-stock stock-disponible';
                    etiqueta.textContent = '🟢 En Stock (' + stockRestante + ')';
                }
            }
        });
    }

    function filtrarProductos() {
        const botonActivo = document.querySelector('.filtro-btn.activo');
        const categoria = botonActivo ? botonActivo.getAttribute('data-categoria') : 'todos';
        const textoBusqueda = inputBuscador.value.toLowerCase().trim();
        const productos = document.querySelectorAll('.producto');

        productos.forEach(producto => {
            const catProducto = producto.getAttribute('data-categoria');
            const nombreProducto = producto.getAttribute('data-nombre').toLowerCase();

            const coincideCategoria = (categoria === 'todos' || catProducto === categoria);
            const coincideBusqueda = nombreProducto.includes(textoBusqueda);

            if (coincideCategoria && coincideBusqueda) {
                producto.style.display = 'flex';
            } else {
                producto.style.display = 'none';
            }
        });
    }

    const botonesFiltro = document.querySelectorAll('.filtro-btn');
    botonesFiltro.forEach(boton => {
        boton.addEventListener('click', (e) => {
            botonesFiltro.forEach(b => b.classList.remove('activo'));
            e.target.classList.add('activo');
            filtrarProductos();
        });
    });

    inputBuscador.addEventListener('keyup', filtrarProductos);
